# Запрос чека (TicketRequest)

| Поле | Наименование | Тип данных | Обязательно | Примечания |
| --- | --- | --- | --- | --- |
| operation | Код операции с чеком | [OperationTypeEnum](../../Общие%20типы%20данных/Операции%20с%20чеком.md) | да |  |
| date_time | Дата и время операции | [DateTime](../../Общие%20типы%20данных/Дата%20и%20время.md) | да |  |
| operator | Оператор-кассир | [Operator](../../Общие%20типы%20данных/Сведения%20об%20операторе-кассире.md) | да |  |
| domain | Вид отрасли | [TicketRequest::Domain](Отраслевые%20реквизиты/Сфера%20деятельности%20чека.md) | да |  |
| items | Список товаров, работ,услуг | List<[TicketRequest::Item](Товары,%20работы,%20услуги/Элемент%20чека.md)> | 1…n |  |
| payments | Список платежей разных типов | List<[TicketRequest::Payment](Товары,%20работы,%20услуги/Оплата%20в%20чеке.md)> | 0…n | Не более одного платежа каждого типа. См также<br>[PaymentTypeEnum](../../Общие%20типы%20данных/Типы%20оплат.md) |
| taxes | Список налогов на сумму чека | List<[TicketRequest::Tax](Товары,%20работы,%20услуги/Налог%20в%20чеке.md)> | 0…n | Налоги на позиции и налог на весь чек являются взаимоисключающими. При формировании чека можно указывать налоги либо для каждого элемента списка товаров, работ,  услуг – items или на всю сумму чека в данном поле с разбивкой по типам налогов. Процент налога – уникальное поле (налог с определённым процентом должен встречаться не более одного раза). |
| amounts | Итоги по чеку | [Amounts](Товары,%20работы,%20услуги/Итоги%20по%20чеку.md) | да |  |
| extension_options | Дополнительные сведения о покупателе | [TicketRequest::ExtensionOptions](Дополнительные%20сервисные%20поля/Дополнительные%20сервисные%20поля.md) | нет |  |
| offline_ticket_<br>number | [OFFLINE](../../Процедуры%20установления%20связи/Работа%20в%20автономном%20режиме.md) номер чека | uint32 | нет | Номер чека, сгенерированный устройством при работе в автономном режиме |
| printed_document_<br>number | Номер печатного документа | uint64 | нет |  |
| printed_ticket | Напечатанный чек, представленный одной строкой | string | нет |  |
| fr_shift_number | Номер текущей смены, считанный с ФР | uint32 | нет | Если поле отсутствует, сервер самостоятельно вычисляет номер смены |
| parent_ticket | Родительский чек | [TicketRequest::ParentTicket](Товары,%20работы,%20услуги/Родительский%20чек%20возврата.md) | нет | Поля содержат<br>данные чека продажи или покупки, на основании которого оформляется<br>возврат.  (*) Обязательно при [OperationTypeEnum](../../Общие%20типы%20данных/Операции%20с%20чеком.md) == OPERATION_BUY_RETURN или [OperationTypeEnum](../../Общие%20типы%20данных/Операции%20с%20чеком.md) == OPERATION_SELL_RETURN |

### Спецификация на уровне Protobuf

Имя файла: [ticket.proto](../../../proto/ticket.proto)

```protobuf
message TicketRequest {
  message Tax {
    required TaxTypeEnum type = 1;
    optional TaxationTypeEnum taxation_type = 2; // optional с версии 200
    required uint32 percent = 3;
    required Money sum = 4;
    required bool is_in_total_sum = 5;
  }

  message Modifier {
    optional string name = 1;
    required Money sum = 2;
    repeated Tax taxes = 3;
    repeated KeyValuePair auxiliary = 4; // добавлено в версии 200
  }

  message Item {

    message Commodity {
      reserved 8;
      reserved "excise_stamp";

      optional uint64 code = 1; // uint32 с версии 200
      optional string name = 2;
      required string section_code = 3;
      required uint64 quantity = 4; // до версии 203 тип uint32
      required Money price = 5;
      required Money sum = 6;
      repeated Tax taxes = 7;
      repeated KeyValuePair auxiliary = 9; // добавлено в версии 200
      optional string physical_label = 10; // добавлено в версии 200
      optional string product_id = 11; // добавлено в версии 200
      optional string barcode = 12; // добавлено в версии 200
      required string measure_unit_code = 13; // добавлено в версии 201
      repeated string list_excise_stamp = 14; // добавлено в версии 203
      optional string ntin = 15; // добавлено в версии 203
    }

    message StornoCommodity {
      reserved 7;
      reserved "excise_stamp";

      optional string name = 1;
      required string section_code = 2;
      required uint64 quantity = 3; // до версии 203 тип uint32
      required Money price = 4;
      required Money sum = 5;
      repeated Tax taxes = 6;
      repeated KeyValuePair auxiliary = 8; // добавлено в версии 200
      optional string physical_label = 9; // добавлено в версии 200
      optional string product_id = 10; // добавлено в версии 200
      optional string barcode = 11; // добавлено в версии 200
      required string measure_unit_code = 12; // добавлено в версии 201
      repeated string list_excise_stamp = 13; // добавлено в версии 203
      optional string ntin = 14; // добавлено в версии 203
    }

    enum ItemTypeEnum {
      ITEM_TYPE_COMMODITY = 1;
      ITEM_TYPE_STORNO_COMMODITY = 2;
      ITEM_TYPE_MARKUP = 3;
      ITEM_TYPE_STORNO_MARKUP = 4;
      ITEM_TYPE_DISCOUNT = 5;
      ITEM_TYPE_STORNO_DISCOUNT = 6;
    }

    required ItemTypeEnum type = 1;
    optional Commodity commodity = 2; // required если type == ITEM_TYPE_COMMODITY
    optional StornoCommodity storno_commodity = 3; // required при type == ITEM_TYPE_STORNO_COMMODITY
    optional Modifier markup = 4; // required при type == ITEM_TYPE_MARKUP
    optional Modifier storno_markup = 5; // required при type == ITEM_TYPE_STORNO_MARKUP
    optional Modifier discount = 6; // required при type == ITEM_TYPE_DISCOUNT
    optional Modifier storno_discount = 7; // required при type == ITEM_TYPE_STORNO_DISCOUNT
  }

  message Domain {
    message Services {
      required string account_number = 1;
    }

    message GasOil {
      optional string correction_number = 1;
      optional Money correction_sum = 2;
      optional string card_number = 3;
    }

    message Taxi {
      required string car_number = 1;
      required bool is_order = 2;
      required Money current_fee = 3;
    }

    message Parking {
      required DateTime begin_time = 1;
      required DateTime end_time = 2;
    }

    required DomainTypeEnum type = 1;
    optional Services services = 2;
    optional GasOil gasoil = 3;
    optional Taxi taxi = 4;
    optional Parking parking  = 5;
  }

  message ExtensionOptions {
    optional string customer_email = 1;
    optional string customer_phone = 2;
    repeated KeyValuePair auxiliary = 3; // добавлено в версии 200
    optional string customer_iin_or_bin = 4; // добавлено в версии 201
  }

  message Payment {
    message CardPaymentFields {
      optional string pos_terminal_id = 1;
      optional string pos_card_type = 2;
      optional uint32 pos_autorization_code = 3;
      optional uint64 pos_rrn = 4; // до версии 203 тип uint32
      optional uint32 pos_receipt_number = 5;
    }

    message MobilePaymentFields {
      optional string qr_type = 1;
      optional string qr_id = 2;
    }

    required PaymentTypeEnum type = 1;
    required Money sum = 2;
    optional CardPaymentFields card_payment_fields = 3; // добавлено в версии 203
    optional MobilePaymentFields mobile_payment_fields = 4; // добавлено в версии 203
  }

  message Amounts {
    required Money total = 1;
    optional Money taken = 2;
    optional Money change = 3;
    optional Modifier markup = 4;
    optional Modifier discount = 5;
  }

  message ParentTicket {
    required string parent_ticket_number = 1;
    required DateTime parent_ticket_data_time = 2;
    required string kgd_kkm_id = 3;
    required Money parent_ticket_total = 4;
    required bool parent_ticket_is_offline = 5;
  }

  reserved 11, 13, 15;
  reserved "printed_document_number_old", "ad_infos", "shift_document_number";

  required OperationTypeEnum operation = 1;
  required DateTime date_time = 2;
  required Operator operator = 3;
  required Domain domain = 4;
  repeated Item items = 5;
  repeated Payment payments = 6;
  repeated Tax taxes = 7;
  required Amounts amounts = 8;
  optional ExtensionOptions extension_options = 9;
  optional uint32 offline_ticket_number = 10;
  optional uint64 printed_document_number = 16; // uint64 с версии 200
  optional string printed_ticket = 12;
  optional uint32 fr_shift_number = 14;
  optional ParentTicket parent_ticket = 17; // добавлено в версии 203 required при operation == OperationTypeEnum == OPERATION_BUY_RETURN или OperationTypeEnum == OPERATION_SELL_RETURN
}
```
